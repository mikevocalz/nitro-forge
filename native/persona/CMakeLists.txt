cmake_minimum_required(VERSION 3.20)
project(NitroForgePersona VERSION 0.1.0 LANGUAGES C CXX)
find_package(Threads REQUIRED)
add_library(nitroforge_persona src/persona.cpp src/persona_c.cpp)
target_compile_features(nitroforge_persona PUBLIC cxx_std_17)
target_include_directories(nitroforge_persona PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(nitroforge_persona PUBLIC Threads::Threads)
set_target_properties(nitroforge_persona PROPERTIES POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(persona_tracking_test tests/tracking_test.cpp)
  target_link_libraries(persona_tracking_test PRIVATE nitroforge_persona)
  add_test(NAME persona_tracking COMMAND persona_tracking_test)
  add_executable(persona_test tests/persona_test.cpp)
  target_link_libraries(persona_test PRIVATE nitroforge_persona)
  add_test(NAME persona_core COMMAND persona_test)
  add_executable(persona_c_consumer tests/c_consumer.c)
  target_link_libraries(persona_c_consumer PRIVATE nitroforge_persona)
  add_test(NAME persona_c_consumer COMMAND persona_c_consumer)
endif()
install(TARGETS nitroforge_persona ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
